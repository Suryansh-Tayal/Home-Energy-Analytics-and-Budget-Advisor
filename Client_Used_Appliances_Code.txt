function renderEntries(){
 let el=$('entries');
 if(!state.entries.length){
  el.className='empty';
  el.textContent='No usage recorded for this day yet.';
  return;
 }
 el.className='table-wrap';
 el.innerHTML=`<table>
 <thead><tr>
 <th>Date</th><th>Appliance</th><th>Approx. hours</th>
 <th>Power</th><th>Energy</th><th>Cost</th><th></th>
 </tr></thead>
 <tbody>${state.entries.map(e=>`
 <tr>
 <td>${e.date}</td>
 <td><b>${e.name}</b></td>
 <td>${e.hours.toFixed(1)} h</td>
 <td>${e.watts} W</td>
 <td>${e.daily.toFixed(3)} kWh</td>
 <td>${money(e.daily*state.tariff)}</td>
 <td><button class="secondary-btn" onclick="removeEntry(${e.id})">Remove</button></td>
 </tr>`).join('')}</tbody></table>
 <p class="hint">Approximate hours × typical appliance wattage. Add the same appliance again if you want to record another entry.</p>`;
}

To show every appliance a client has ever used (unique names across all dates), the backend response and frontend rendering would need an additional all-history list.
