void saveCustom(const string &u){

  f<<a.name<<","<<a.category<<","<<a.watts<<","<<a.standby<<","<<a.efficiency<<","<<a.priority<<","<<a.status<<"\n";

UserData &loadUser(const string &u){
 if(users.count(u))return users[u];
 UserData d;
 string dir=userDir(u);
 fs::create_directories(dir);
 { ifstream f(dir+"/settings.txt"); string s; if(f>>s)d.solar=(s=="yes"); }
 { ifstream f(dir+"/custom_appliances.csv"); string line; getline(f,line);
   while(getline(f,line)){
     if(trim(line).empty())continue;
     auto v=splitCsv(line,7);
     Appliance a;
     a.id=1000+(int)d.custom.size();
     a.name=v[0];
     a.category=v[1];
     a.watts=atoi(v[2].c_str());
     a.standby=atoi(v[3].c_str());
     a.efficiency=v[4];
     a.priority=v[5];
     a.status=v[6].empty()?"ON":v[6];
     if(!a.name.empty()&&a.watts>0)d.custom.push_back(a);
   }
 }
 { ifstream f(dir+"/usage.csv"); string line; getline(f,line);
   while(getline(f,line)){
     if(trim(line).empty())continue;
     auto v=splitCsv(line,6);
     Usage x;
     x.id=atoi(v[0].c_str()); x.date=v[1]; x.name=v[2];
     x.hours=atof(v[3].c_str()); x.watts=atoi(v[4].c_str());
     if(!validDate(x.date)||x.name.empty()||x.watts<=0||x.hours<=0||x.hours>24)continue;
     x.daily=x.watts*x.hours/1000.0;
     d.usages.push_back(x);
     d.nextId=max(d.nextId,x.id+1);
   }
 }
 users[u]=d;
 return users[u];
}

vector<Appliance> catalogueFor(const string &u){
 vector<Appliance> c=DEFAULT_APPLIANCES;

 return c;
}

function openCustom(){
 $('customBackdrop').classList.add('show');
 $('customMsg').textContent='';
}
function closeCustom(){
 $('customBackdrop').classList.remove('show');
}
async function saveCustom(e){
 e.preventDefault();
 let f=new FormData($('customForm'));
 let data=Object.fromEntries(f.entries());
 try{
  await api('/api/custom-appliance',data);
  $('customMsg').textContent='Saved! Appliance added to your list.';
  await refresh();
  setTimeout(()=>{
   closeCustom();
   $('customForm').reset();
  },650);
 }catch(err){
  $('customMsg').textContent=err.message;
 }
}

============================================================
FEATURE WORKFLOW
============================================================

1. User clicks “Can’t find your appliance? Add it yourself”.
2. A popup form opens.
3. User enters appliance name, category, rated power, standby power, efficiency, priority and status.
4. JavaScript sends the form data to /api/custom-appliance.
5. The C++ backend validates the details and checks for duplicate names.
6. The appliance is added to that user's custom appliance list.
7. saveCustom() stores it in that user's custom_appliances.csv file.
8. The custom appliance is loaded again after restart and added to the appliance catalogue.

NOTE: This file contains the feature's relevant code sections from the supplied project ZIP. It is a reference/extract, not a standalone compilable program. Existing project helper functions such as userDir(), trim(), splitCsv(), parseForm(), cleanField(), lowerText(), api() and refresh() are used by these sections.
